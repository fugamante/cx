\section*{Purpose}
This playbook is a set of reproducible procedures. It is intentionally low-prose and high-signal.

\section{P0: Confirm Runtime Path}
\begin{lstlisting}[style=cx]
./bin/xshelf version
./bin/xshelf where version
\end{lstlisting}

\section{P1: Confirm Schemas Exist}
\begin{lstlisting}[style=cx]
./bin/xshelf schema list
ls .cx/schemas
\end{lstlisting}

\section{P2: Validate Structured Output (Schema Commands)}
\subsection*{Commit JSON}
\begin{lstlisting}[style=cx]
./bin/xshelf commitjson | jq .
\end{lstlisting}

\subsection*{Diff Summary}
\begin{lstlisting}[style=cx]
./bin/xshelf diffsum-staged | jq .
\end{lstlisting}

\subsection*{Next Commands}
\begin{lstlisting}[style=cx]
./bin/xshelf next git status | jq .
\end{lstlisting}

\section{P3: Budget Stress Test}
\begin{lstlisting}[style=cx]
CX_CONTEXT_BUDGET_CHARS=2000 CX_CONTEXT_BUDGET_LINES=40 ./bin/xshelf cxo git status
./bin/xshelf budget
./bin/xshelf trace
\end{lstlisting}

\section{P4: Quarantine and Replay (Failure Procedure)}
\begin{lstlisting}[style=cx]
./bin/xshelf quarantine list
./bin/xshelf quarantine show <id>
./bin/xshelf replay <id>
\end{lstlisting}

\section{P5: Safety Gate (Before Any Execution)}
\begin{lstlisting}[style=cx]
./bin/xshelf policy show
./bin/xshelf policy check "<candidate command>"
\end{lstlisting}

\section{P6: Task Graph (Orchestration)}
\subsection*{Create and Execute}
\begin{lstlisting}[style=cx]
./bin/xshelf task add "Objective: harden capture clipping" --role implementer
./bin/xshelf task list --status pending
./bin/xshelf task run-all --status pending
\end{lstlisting}

\subsection*{Fan-out}
\begin{lstlisting}[style=cx]
./bin/xshelf task fanout "Objective: refactor logging" --from staged-diff
./bin/xshelf task list --status pending
\end{lstlisting}

\section{P7: Telemetry Contract Validation}
\begin{lstlisting}[style=cx]
./bin/xshelf logs validate --fix=false
tail -n 1 .cx/cxlogs/runs.jsonl | jq .
\end{lstlisting}

\section{P8: Self-Optimization}
\begin{lstlisting}[style=cx]
./bin/xshelf optimize 200
./bin/xshelf optimize 200 --json | jq .
\end{lstlisting}
